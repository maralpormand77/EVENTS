تنظیمات دیتابیس متمرکز PostgreSQL (Supabase)

با اتصال به دیتابیس واقعی PostgreSQL (از طریق Supabase)، تمام ثبت‌نام‌ها به شکل متمرکز ذخیره شده و بلادرنگ در پنل ادمین و بین تمام دستگاه‌ها و پرسنل همگام می‌شوند.

راهنمای سریع راه‌اندازی در ۳ گام (رایگان در ۲ دقیقه):
  1. وارد سایت supabase.com شده، یک حساب کاربری بسازید و دکمه New Project را بزنید.
  2. از منوی سمت چپ وارد SQL Editor شوید، کد SQL زیر را پیست کرده و دکمه سبز Run را بزنید:
-- ایجاد جدول ثبت‌نام رویدادها در PostgreSQL
CREATE TABLE IF NOT EXISTS registrations (
  id TEXT PRIMARY KEY,
  event_id TEXT NOT NULL,
  event_title TEXT NOT NULL,
  personnel_code TEXT NOT NULL,
  full_name TEXT NOT NULL,
  status TEXT NOT NULL,
  status_text TEXT,
  timestamp TEXT,
  jalali_date TEXT,
  user_agent TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE registrations ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public insert" ON registrations FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow public read" ON registrations FOR SELECT USING (true);
CREATE POLICY "Allow public delete" ON registrations FOR DELETE USING (true);
